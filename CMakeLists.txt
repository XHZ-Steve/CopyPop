cmake_minimum_required(VERSION 3.25)
project(CopyPop VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "CopyPop requires Windows 10 22H2 or Windows 11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Network Svg Test)
qt_standard_project_setup()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
qt_add_library(copypop_core STATIC src/core/Config.cpp src/core/Config.h src/platform/Windows.cpp src/platform/Windows.h)
target_include_directories(copypop_core PUBLIC src)
target_link_libraries(copypop_core PUBLIC Qt6::Gui Qt6::Network user32 shell32 ole32 oleaut32 uuid dwmapi)
qt_add_executable(CopyPop WIN32 src/main.cpp src/App.cpp src/App.h src/platform/SelectionTracker.cpp src/platform/SelectionTracker.h src/render/OverlayRenderer.cpp src/render/OverlayRenderer.h resources/copypop.rc)
target_include_directories(CopyPop PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/resources")
target_link_libraries(CopyPop PRIVATE copypop_core Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
file(GLOB QML_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "qml/*.qml")
file(GLOB ICON_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "resources/icons/*.svg" "resources/data/*.json")
qt_add_resources(CopyPop assets PREFIX "/" FILES ${QML_FILES} ${ICON_FILES})
if(MSVC)
  target_compile_options(copypop_core PRIVATE /W4 /utf-8)
  target_compile_options(CopyPop PRIVATE /W4 /utf-8)
else()
  target_compile_options(copypop_core PRIVATE -Wall -Wextra)
  target_compile_options(CopyPop PRIVATE -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE copypop_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  qt_add_executable(ui_tests tests/ui_tests.cpp src/App.cpp src/App.h src/platform/SelectionTracker.cpp src/platform/SelectionTracker.h src/render/OverlayRenderer.cpp src/render/OverlayRenderer.h)
  target_link_libraries(ui_tests PRIVATE copypop_core Qt6::Test Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
  if(MSVC)
    target_compile_options(ui_tests PRIVATE /utf-8)
  endif()
  set_target_properties(core_tests ui_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  qt_add_resources(ui_tests test_assets PREFIX "/" FILES ${QML_FILES} ${ICON_FILES})
  qt_add_executable(uia_fixture WIN32 tests/uia_fixture.cpp)
  target_link_libraries(uia_fixture PRIVATE user32 gdi32)
  add_dependencies(ui_tests uia_fixture)
  add_test(NAME ui_native COMMAND ui_tests)
  set_tests_properties(ui_native PROPERTIES TIMEOUT 60)
endif()
install(TARGETS CopyPop RUNTIME DESTINATION .)
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
set(CPACK_PACKAGE_NAME "CopyPop")
set(CPACK_PACKAGE_VENDOR "CopyPop contributors")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "CopyPop")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_NSIS_DISPLAY_NAME "CopyPop")
set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
set(CPACK_PACKAGE_EXECUTABLES "CopyPop;CopyPop")
set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
set(CPACK_VERBATIM_VARIABLES YES)
set(CPACK_NSIS_EXTRA_UNINSTALL_COMMANDS "DeleteRegValue HKCU 'Software\\Microsoft\\Windows\\CurrentVersion\\Run' 'CopyPop'")
include(CPack)
